```text
python3 scripts/validate_rehearsal.py --phase agents
npx --yes hyperframes@0.8.4 lint .
npx --yes hyperframes@0.8.4 check .
npx --yes hyperframes@0.8.4 preview .
npx --yes hyperframes@0.8.4 snapshot . --time <shot-time> --output <path>
npx --yes hyperframes@0.8.4 render . --output renders/BC-REHEARSAL-001-directed-animatic-final.mp4 --fps 30 --quality standard --workers auto --strict-all --no-best-effort --skill general-video
ffprobe -v error -show_format -show_streams hyperframes/renders/BC-REHEARSAL-001-directed-animatic-final.mp4
ffmpeg -v info -i hyperframes/renders/BC-REHEARSAL-001-directed-animatic-final.mp4 -vf blackdetect=d=0.1 -an -f null -
python3 scripts/validate_rehearsal.py --phase build
python3 scripts/validate_rehearsal.py --phase final
```
